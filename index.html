Replace the sample values with numbers exported from your Tadoba explorer script.

  4. Paste it into the box and choose Apply data. Every chart on the page updates.
Earth Engine export snippet 
// Tadoba web page: export chart data from Earth Engine
// Paste at the END of the Tadoba explorer script and click Run.
// Copy the JSON line printed in the Console into the web page.

var YEAR = 2024;
var start = ee.Date.fromYMD(YEAR, 1, 1);
var months = ee.List.sequence(0, 11);

// Mean of a band over the study area for each month
function monthly(col, band, useSum, scale) {
  return months.map(function (m) {
    var s = start.advance(ee.Number(m), 'month');
    var sub = col.filterDate(s, s.advance(1, 'month')).select(band);
    var img = ee.Image(ee.Algorithms.If(
      sub.size().gt(0),
      useSum ? sub.sum() : sub.mean(),
      ee.Image.constant(0).float().rename(band).updateMask(0)
    ));
    return img.reduceRegion({
      reducer: ee.Reducer.mean(), geometry: aoi, scale: scale,
      bestEffort: true, maxPixels: 1e10
    });
  });
}

var rainCol = ee.ImageCollection('NASA/GPM_L3/IMERG_MONTHLY_V07')
  .select('precipitation')
  .map(function (img) {
    var hours = img.date().advance(1, 'month').difference(img.date(), 'hour');
    return img.multiply(hours).toFloat().rename('rain')
      .copyProperties(img, ['system:time_start']);
  });

var landsat = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .merge(ee.ImageCollection('LANDSAT/LC09/C02/T1_L2'))
  .filterBounds(aoi)
  .filter(ee.Filter.lt('CLOUD_COVER', 80))
  .map(maskLandsat);

var ndviCol = landsat.map(function (img) {
  return img.select(['SR_B4', 'SR_B5']).multiply(0.0000275).add(-0.2)
    .normalizedDifference(['SR_B5', 'SR_B4']).toFloat().rename('ndvi')
    .copyProperties(img, ['system:time_start']);
});

var lstCol = landsat.map(function (img) {
  return img.select('ST_B10').multiply(0.00341802).add(149.0).subtract(273.15)
    .toFloat().rename('lst')
    .copyProperties(img, ['system:time_start']);
});

var smCol = ee.ImageCollection('NASA/SMAP/SPL3SMP_E/006').map(function (img) {
  var good = img.select('retrieval_qual_flag_am').toInt().bitwiseAnd(1).eq(0);
  return img.select('soil_moisture_am').toFloat().rename('sm').updateMask(good)
    .copyProperties(img, ['system:time_start']);
});

// Class maps for the pie charts
var d = getData(YEAR);
var heightClass = d.image.select('rh98').divide(5).floor().min(5).toInt().rename('c');
var slopeClass = slope.gt(5).add(slope.gt(10)).add(slope.gt(20)).add(slope.gt(30)).rename('c');
var landCover = ee.ImageCollection('ESA/WorldCover/v200').first().select('Map').rename('c');

function hist(img, scale) {
  return img.clip(aoi).reduceRegion({
    reducer: ee.Reducer.frequencyHistogram(), geometry: aoi, scale: scale,
    bestEffort: true, maxPixels: 1e10
  }).get('c');
}

ee.Dictionary({
  rain: monthly(rainCol, 'rain', true, 11000),
  ndvi: monthly(ndviCol, 'ndvi', false, 100),
  lst: monthly(lstCol, 'lst', false, 100),
  sm: monthly(smCol, 'sm', false, 9000),
  landCover: hist(landCover, 30),
  heightClasses: hist(heightClass, 25),
  slopeClasses: hist(slopeClass, 30)
}).evaluate(function (r, err) {
  if (err) { print('Export failed: ' + err); return; }
  function flat(list, band) {
    return list.map(function (x) {
      var v = x[band];
      return (v === null || v === undefined) ? null : Math.round(v * 1000) / 1000;
    });
  }
  print(JSON.stringify({
    year: YEAR,
    monthly: {
      rain: flat(r.rain, 'rain'), ndvi: flat(r.ndvi, 'ndvi'),
      lst: flat(r.lst, 'lst'), sm: flat(r.sm, 'sm')
    },
    landCover: r.landCover || {},
    heightClasses: r.heightClasses || {},
    slopeClasses: r.slopeClasses || {}
  }));
});

Datasets: GPM IMERG V07 (rainfall), Landsat 8 and 9 Collection 2 (NDVI, land surface temperature), SMAP L3 enhanced (soil moisture), GEDI L2A monthly raster (canopy height), SRTM (slope), ESA WorldCover 2021 (land cover).
